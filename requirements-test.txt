# NLP extraction tests exercise the real pipeline instead of a stub. The model

#   python -m spacy download en_core_web_sm
# Tests that need it skip cleanly when it is absent (see
# tests/unit/test_nlp_extract_spacy.py), so this file stays usable without it.

click>=8.0
